type State = Record<string, any>;
type Init = unknown;
type Output = unknown;
type Context = Record<string, unknown>;
type Inference = {
  type: 'output' | 'tool_call';
  data: Record<string, unknown>;
};

const lifecyle = {
  onInit: (_event: Init): Context => ({}),
  // biome-ignore lint/suspicious/noExplicitAny: Stuff
  onResume: (_prevContext: Context, _toolResponses: any): Context => ({}),
  preInference: (_context: Context): Context => ({}),
  infer: (_context: Context): Inference => ({}) as unknown as Inference,
  postInference: (_inference: Inference): Inference => ({}) as unknown as Inference,
  onComplete: (_inference: Inference): { error: Error } | { output: Output } => {
    return { output: {} };
  },
  // biome-ignore lint/suspicious/noExplicitAny: Stuff
  toolCallPermission: (data: any) => {}, 
  preToolCall: (
    _id: string,
    _source: string,
    _type: string,
    // biome-ignore lint/suspicious/noExplicitAny: Stuff
    _data: Record<string, any>,
    // biome-ignore lint/suspicious/noExplicitAny: Stuff
    _metadata: Record<string, any>,
  ): ({ action: 'DROP' | 'CONTINUE' } | {error: Error}) => ({
    error: new Error('')
  }),
};

const inference = (_context: Context): Inference => {
  const context = lifecyle.preInference(_context);
  const resp = lifecyle.infer(context);
  return lifecyle.postInference(resp);
};
